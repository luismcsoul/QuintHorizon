def normalize_name(name):
    if pd.isna(name) or not str(name).strip(): return None
    return re.sub(r'\s+', ' ', str(name).strip())


def canonical_name(name):
    n = normalize_name(name)
    return n.casefold() if n else None


def is_forbidden_name(name):
    c = canonical_name(name)
    return c is None or c in FORBIDDEN_NAMES


def bce_to_jd(bce_or_ce_year, month=12, day=1, hour=0):
    year = bce_or_ce_year + 1 if bce_or_ce_year < 0 else bce_or_ce_year
    if month <= 2:
        year -= 1
        month += 12
    A = np.floor(year / 100)
    B = 2 - A + np.floor(A / 4)
    return np.floor(365.25 * (year + 4716)) + np.floor(30.6001 * (month + 1)) + day + (hour / 24.0) - 1524.5


def rozenberg_1966_airmass(alt_deg):
    if alt_deg <= -1.5: return 99.0
    z_deg = 90.0 - alt_deg
    z_rad = np.radians(max(z_deg, 0.0))
    return 1.0 / (np.cos(z_rad) + 0.025 * np.exp(-11.0 * np.cos(z_rad)))


def schaefer_1993_extinction(alt_deg, pressure_hpa, site_kv):
    X = rozenberg_1966_airmass(alt_deg)
    if X >= 90.0: return 99.0
    p_ratio = pressure_hpa / 1013.25
    k_rayleigh = 0.1451 * p_ratio
    k_aerosol = max(0.01, site_kv - k_rayleigh)
    return (k_rayleigh + k_aerosol) * X


def assign_sector(dec, delta_ra):
    abs_dec = abs(dec)
    is_east = delta_ra <= 0
    is_north = dec > 0
    if abs_dec <= EQUATOR_LIMIT: return "EAST" if is_east else "WEST"
    elif abs_dec >= POLAR_LIMIT: return "NORTH" if is_north else "SOUTH"
    else:
        if is_north and is_east: return "NORTHEAST"
        elif is_north and not is_east: return "NORTHWEST"
        elif not is_north and not is_east: return "SOUTHWEST"
        else: return "SOUTHEAST"


def angular_distance_from_midpoint(ra, midpoint):
    return ((ra - midpoint + 180.0) % 360.0) - 180.0


def score_individual_candidates(df):
    if df.empty: return df.copy()
    df = df.copy()
    df["IQ_DEC_SCORE"] = (1.0 - (df["IQ_DEC"].abs() / 30.0)).clip(0.0, 1.0)
    v_min, v_max = df["IQ_V_APP"].min(), df["IQ_V_APP"].max()
    df["IQ_BRIGHTNESS_SCORE"] = (1.0 - ((df["IQ_V_APP"] - v_min) / (v_max - v_min))) if v_max > v_min else 1.0
    return df


def safe_get_col(df, candidates):
    col_lookup = {col.lower(): col for col in df.columns}
    for cand in candidates:
        if cand.lower() in col_lookup:
            actual_col = col_lookup[cand.lower()]
            return df[actual_col].values
    raise KeyError(f"None of {candidates} found in DataFrame columns: {list(df.columns)}")


def generate_emergence_pool(df, magnitude_limit):
    v_app_emergence = safe_get_col(df, ["V_App_Emergence", "IQ_V_APP", "V_App_Culm"])
    max_alt = safe_get_col(df, ["Max_Alt"])
    mask = (v_app_emergence <= magnitude_limit) & (max_alt > 0.0)
    return df[mask].copy()

# =====================================================================


def run_epoch(epoch_key, epoch_index, base_stars_df):
    site_cfg = EPOCH_DB[epoch_key]
    year = site_cfg["year"]
    year_str = f"{abs(year)} BCE" if year < 0 else f"{year} CE"


    print(f"Processing Epoch {epoch_index}/7: {site_cfg['site_name']} ({year_str})...")


    jd = bce_to_jd(year, hour=0)
    t = Time(jd, format='jd', scale='utc')
    fk5_epoch = FK5(equinox=t)
    h_pyramid = site_cfg["platform_height"]
    h_total = site_cfg["elevation_asl"] + h_pyramid
    terrain_mask = site_cfg["terrain_mask_deg"]


    horizon_dip_deg = 0.0347 * np.sqrt(h_pyramid)
    effective_horizon_alt = max(terrain_mask, -horizon_dip_deg)
    location = EarthLocation(lat=site_cfg["lat"]*u.deg, lon=site_cfg["lon"]*u.deg, height=h_total*u.m)
    times_24h = t + np.linspace(0, 1, 720) * u.day
    altaz_frame = AltAz(obstime=times_24h, location=location, pressure=site_cfg["pressure_hpa"]*u.hPa, temperature=site_cfg["temp_c"]*u.deg_C)


    mintaka_icrs = SkyCoord(ra="05h32m00.40s", dec="-00d17m56.7s", pm_ra_cosdec=0.59*u.mas/u.yr, pm_dec=-0.64*u.mas/u.yr, frame='icrs')
    mintaka_epoch = mintaka_icrs.transform_to(fk5_epoch)
    mintaka_ra_epoch = mintaka_epoch.ra.deg
    dynamic_framework_offset = (mintaka_ra_epoch - HALF_WIDTH) % 360.0


    stars = base_stars_df.copy()
    epoch_coords, max_alts, v_apps_culm, h_vis_list, rise_azimuths, status_list = [], [], [], [], [], []


    for _, row in stars.iterrows():
        plx_mas = row["Plx"]
        dist_pc = (1000.0 / plx_mas) if (pd.notnull(plx_mas) and plx_mas > 0) else None
        coord_kwargs = {'ra': row["RAICRS"] * u.deg, 'dec': row["DEICRS"] * u.deg, 'pm_ra_cosdec': row["pmRA"] * u.mas / u.yr, 'pm_dec': row["pmDE"] * u.mas / u.yr, 'frame': 'icrs'}
        if dist_pc: coord_kwargs['distance'] = dist_pc * u.pc
        coord = SkyCoord(**coord_kwargs).transform_to(fk5_epoch)
        epoch_coords.append((coord.ra.deg, coord.dec.deg))


        star_altaz = coord.transform_to(altaz_frame)
        alts, azs = star_altaz.alt.deg, star_altaz.az.deg
        max_idx, min_idx = np.argmax(alts), np.argmin(alts)
        max_alt_app = alts[max_idx] + horizon_dip_deg
        min_alt_app = alts[min_idx] + horizon_dip_deg
        ext_culm = schaefer_1993_extinction(max_alt_app, site_cfg["pressure_hpa"], site_cfg["k_v"])
        v_app_culm = row["Vmag"] + ext_culm if max_alt_app > 0 else 99.0


        alts_48 = np.concatenate((alts, alts))
        azs_48 = np.concatenate((azs, azs))
        d_alt_48 = np.diff(alts_48, prepend=alts_48[0])
        rising_nodes = np.where((alts_48[:720] < effective_horizon_alt) & (alts_48[1:721] >= effective_horizon_alt) & (d_alt_48[1:721] > 0))[0]


        true_rise_az, first_visible_az, first_visible_h = None, None, None
        if len(rising_nodes) > 0:
            idx = rising_nodes[0]
            h1, h2 = alts_48[idx], alts_48[idx + 1]
            az1, az2 = azs_48[idx], azs_48[idx + 1]
            if abs(az2 - az1) > 180: az2 += 360 if az1 > az2 else -360
            frac = (effective_horizon_alt - h1) / (h2 - h1) if (h2 - h1) != 0 else 0
            true_rise_az = (az1 + frac * (az2 - az1)) % 360.0


            max_seen_alt = -999.0
            for alt_i, az_i in zip(alts_48[idx : idx + 720], azs_48[idx : idx + 720]):
                if alt_i < max_seen_alt: break
                max_seen_alt = alt_i
                ext_i = schaefer_1993_extinction(alt_i + horizon_dip_deg, site_cfg["pressure_hpa"], site_cfg["k_v"])
                if (row["Vmag"] + ext_i) <= NAKED_EYE_LIMIT_MAG:
                    first_visible_az, first_visible_h = az_i, alt_i
                    break


        if min_alt_app > 0:
            az_str, h_vis_str, status = "CIRCUMPOLAR", "N/A", "CIRCUMPOLAR"
        elif true_rise_az is not None and first_visible_az is not None:
            az_str, h_vis_str, status = f"{true_rise_az:.2f}°", f"{first_visible_h:.1f}°", "VISIBLE BEACON"
        elif max_alt_app > 0:
            az_str, h_vis_str = f"~{azs[max_idx]:.2f}° (Culm)", "N/A"
            status = "EXTINGUISHED" if v_app_culm > NAKED_EYE_LIMIT_MAG else "LOW CULMINATION"
        else:
            az_str, h_vis_str, status = "Never Rises", "N/A", "SUB-HORIZON"


        max_alts.append(max_alt_app)
        v_apps_culm.append(v_app_culm)
        h_vis_list.append(h_vis_str)
        rise_azimuths.append(az_str)
        status_list.append(status)


    coords_df = pd.DataFrame(epoch_coords, columns=["RA_EPOCH", "DEC_EPOCH"])
    stars["RA_EPOCH"], stars["DEC_EPOCH"] = coords_df["RA_EPOCH"], coords_df["DEC_EPOCH"]
    stars["Max_Alt"], stars["V_App_Culm"] = max_alts, v_apps_culm
    stars["H_Vis"], stars["Rise_Azimuth"], stars["Status"] = h_vis_list, rise_azimuths, status_list


    stars["RA_FRAME"] = (stars["RA_EPOCH"] - dynamic_framework_offset) % 360
    stars["Quintant"] = (stars["RA_FRAME"] // QUINTANT_WIDTH).astype(int) + 1
    stars["Center_RA"] = (stars["Quintant"] - 1) * QUINTANT_WIDTH + HALF_WIDTH
    stars["Delta_RA"] = (stars["RA_FRAME"] - stars["Center_RA"] + 180) % 360 - 180
    stars["Sector"] = stars.apply(lambda r: assign_sector(r["DEC_EPOCH"], r["Delta_RA"]), axis=1)
    visible_stars = stars[stars["Status"].isin(["VISIBLE BEACON", "CIRCUMPOLAR"])].copy()


    # --- Extract Active Primary Anchors & Candidate Pool ---
    final_anchor_rows = []
    candidate_pool_rows = []
    for q in range(1, 6):
        q_stars = visible_stars[visible_stars["Quintant"] == q]
        for sec in CUSTOM_SECTOR_ORDER:
            sec_stars = q_stars[q_stars["Sector"] == sec].sort_values(by="Vmag")
            if not sec_stars.empty:
                candidate_pool_rows.extend(sec_stars.head(MAX_STARS_PER_SECTOR).to_dict('records'))
                top_star = sec_stars.iloc[0]
                final_anchor_rows.append(top_star)


    final_anchors_df = pd.DataFrame(final_anchor_rows)
    candidate_pool_df = pd.DataFrame(candidate_pool_rows)


    if SORT_BY.lower() == 'declination':
        final_anchors_df = final_anchors_df.sort_values(by="DEC_EPOCH", ascending=False)
    else:
        sector_rank = {sec: i for i, sec in enumerate(CUSTOM_SECTOR_ORDER)}
        final_anchors_df["Sector_Rank"] = final_anchors_df["Sector"].map(sector_rank)
        final_anchors_df = final_anchors_df.sort_values(by=["Quintant", "Sector_Rank"])


    # --- Inter-Quintant Candidates ---
    revalidation_pool = visible_stars.copy()
    revalidation_pool["IQ_DEC"] = pd.to_numeric(revalidation_pool["DEC_EPOCH"], errors="coerce")
    revalidation_pool["IQ_RA_FRAME"] = pd.to_numeric(revalidation_pool["RA_FRAME"], errors="coerce")
    revalidation_pool["IQ_VMAG"] = pd.to_numeric(revalidation_pool["Vmag"], errors="coerce")
    revalidation_pool["IQ_V_APP"] = pd.to_numeric(revalidation_pool["V_App_Culm"], errors="coerce").fillna(revalidation_pool["IQ_VMAG"])
    revalidation_pool = revalidation_pool.dropna(subset=["IQ_DEC", "IQ_RA_FRAME", "IQ_V_APP"]).copy()


    inter_quintant_rows = []
    for central_q in range(1, 6):
        previous_q, upcoming_q = ((central_q - 2) % 5) + 1, (central_q % 5) + 1
        central_midpoint = (central_q - 1) * QUINTANT_WIDTH + HALF_WIDTH
        q_pool = revalidation_pool.copy()
        q_pool["IQ_DELTA_RA"] = q_pool["IQ_RA_FRAME"].apply(lambda x: angular_distance_from_midpoint(x, central_midpoint))


        central_candidates = q_pool[q_pool["IQ_DEC"].abs() <= 30.0].copy()
        central_candidates["IQ_ABS_DELTA_RA"] = central_candidates["IQ_DELTA_RA"].abs()
        central_selected = central_candidates.sort_values(by=["IQ_ABS_DELTA_RA", "IQ_V_APP"]).iloc[0] if not central_candidates.empty else None


        west_pool = q_pool[(q_pool["Quintant"] == previous_q) & (q_pool["IQ_DELTA_RA"] < 0) & (q_pool["IQ_DELTA_RA"] >= -80.0)].copy()
        west_primary = west_pool[west_pool["IQ_DEC"].abs() <= 16.0].copy()
        west_candidates = score_individual_candidates(west_primary if not west_primary.empty else west_pool[west_pool["IQ_DEC"].abs() <= 30.0].copy())


        east_pool = q_pool[(q_pool["Quintant"] == upcoming_q) & (q_pool["IQ_DELTA_RA"] > 0) & (q_pool["IQ_DELTA_RA"] <= 80.0)].copy()
        east_primary = east_pool[east_pool["IQ_DEC"].abs() <= 16.0].copy()
        east_candidates = score_individual_candidates(east_primary if not east_primary.empty else east_pool[east_pool["IQ_DEC"].abs() <= 30.0].copy())


        pair_results = []
        if not west_candidates.empty and not east_candidates.empty:
            for _, west in west_candidates.iterrows():
                for _, east in east_candidates.iterrows():
                    sym_error = abs(abs(west["IQ_DELTA_RA"]) - abs(east["IQ_DELTA_RA"]))
                    sym_score = max(0.0, min(1.0, 1.0 - (sym_error / 80.0)))
                    pair_bright = (west["IQ_BRIGHTNESS_SCORE"] + east["IQ_BRIGHTNESS_SCORE"]) / 2.0
                    pair_dec = (west["IQ_DEC_SCORE"] + east["IQ_DEC_SCORE"]) / 2.0
                    pair_score = (0.40 * pair_bright) + (0.35 * sym_score) + (0.25 * pair_dec)
                    pair_results.append({
                        "West": west, "East": east, "West_Delta": west["IQ_DELTA_RA"], "East_Delta": east["IQ_DELTA_RA"],
                        "Symmetry_Error": sym_error, "Symmetry_Score": sym_score,
                        "Pair_Brightness": pair_bright, "Pair_DEC": pair_dec, "Pair_Score": pair_score
                    })


        best_pair = sorted(pair_results, key=lambda x: (x["Pair_Score"], x["Symmetry_Score"], x["Pair_Brightness"]), reverse=True)[0] if pair_results else None
        west_selected, east_selected = (best_pair["West"], best_pair["East"]) if best_pair else (None, None)


        inter_quintant_rows.append({
            "Central_Quintant": central_q,
            "Previous_Quintant": previous_q,
            "Upcoming_Quintant": upcoming_q,
            "Central_Midpoint": central_midpoint,
            "Central_Reference": central_selected["Star_Name"] if central_selected is not None else "NONE",
            "Central_Delta": central_selected["IQ_DELTA_RA"] if central_selected is not None else np.nan,
            "West": west_selected["Star_Name"] if west_selected is not None else "NONE",
            "West_Delta": west_selected["IQ_DELTA_RA"] if west_selected is not None else np.nan,
            "East": east_selected["Star_Name"] if east_selected is not None else "NONE",
            "East_Delta": east_selected["IQ_DELTA_RA"] if east_selected is not None else np.nan,
            "Symmetry_Error": best_pair["Symmetry_Error"] if best_pair is not None else np.nan,
            "Pair_Score": best_pair["Pair_Score"] if best_pair is not None else np.nan
        })
    inter_quintant_df = pd.DataFrame(inter_quintant_rows)


    # --- Meridional Candidates ---
    ns_candidates = candidate_pool_df.copy()
    ns_candidates["NS_V_APP"] = pd.to_numeric(ns_candidates["V_App_Culm"], errors="coerce")
    ns_candidates["NS_DEC"] = pd.to_numeric(ns_candidates["DEC_EPOCH"], errors="coerce")
    ns_candidates["NS_RA"] = pd.to_numeric(ns_candidates["RA_EPOCH"], errors="coerce")
    ns_candidates = ns_candidates.dropna(subset=["NS_V_APP", "NS_DEC", "NS_RA"]).copy()

                    if len(w_idx) > 0 and len(e_idx) > 0:
                        w_primary_idx = w_idx[np.abs(t_dec[w_idx]) <= 16.0]
                        w_pool_idx = w_primary_idx if len(w_primary_idx) > 0 else w_idx[np.abs(t_dec[w_idx]) <= 30.0]

                        e_pool_idx = e_primary_idx if len(e_primary_idx) > 0 else e_idx[np.abs(t_dec[e_idx]) <= 30.0]

                            for e in e_pool_idx:
                                sym_score = max(0.0, min(1.0, 1.0 - (abs(abs(sim_delta_ra[w]) - abs(sim_delta_ra[e])) / 80.0)))
                                geom_score = (0.5 * sym_score) + (0.5 * ((t_dec_scores[w] + t_dec_scores[e]) / 2.0))
                                if geom_score > best_geom_score:
                                    best_geom_score, sim_w_name, sim_e_name, best_pair_vmag = geom_score, t_names[w], t_names[e], (t_vapp[w] + t_vapp[e]) / 2.0


                    if best_geom_score > 0:
                        pure_geom_score += best_geom_score
                        if best_pair_vmag is not None: selected_v_mags.append(best_pair_vmag)
                    sequence.extend([sim_w_name, sim_e_name, sim_n_name, sim_s_name])


                return sequence, pure_geom_score, np.mean(selected_v_mags) if len(selected_v_mags) > 0 else np.nan

def run_20_selector(panoramic_df, inter_df, meridional_df, n_epochs):
    print("\n" + "=" * 110)
    print("20-STAR INTEGRATED SELECTOR: PHASE 1 (E/W) + PHASE 2 (N/S) [BRIGHTNESS ONLY]")
    print("=" * 110)


    brightness_lookup = defaultdict(list)
    for _, row in panoramic_df.iterrows():
        brightness_lookup[(int(row["Epoch"]), row["StarKey"])].append(float(row["V_App_Culm"]))


    def best_vapp_for_star(star_key):
        vals = [v for (ep, sk), vlist in brightness_lookup.items() if sk == star_key for v in vlist]
        return float(min(vals)) if vals else np.nan


    def brightness_score(vapp):
        return float(np.clip((VAPP_MAX - float(vapp)) / VAPP_MAX, 0.0, 1.0)) if not pd.isna(vapp) else 0.0


    # Phase 1: EAST/WEST LOCK
    p1_inter_best = {}
    for _, row in inter_df.iterrows():
        key = (row["Final_Quintant"], row["Cardinal"], row["StarKey"])
        if key not in p1_inter_best or row["Score"] > p1_inter_best[key]["Score"]:
            p1_inter_best[key] = row.to_dict()


    p1_general_pool = defaultdict(dict)
    for _, row in panoramic_df.iterrows():
        q, sec, sk = int(row["Quintant"]), row["Sector"], row["StarKey"]
        for card, smap in SECTOR_FALLBACK.items():
            if sec in smap:
                key = (q, card, sk)
                cand = {"Star": row["Star"], "V_App": float(row["V_App_Culm"])}
                if key not in p1_general_pool or cand["V_App"] < p1_general_pool[key]["V_App"]:
                    p1_general_pool[key] = cand


    QUINTANTS = [1, 2, 3, 4, 5]
    CARDINALS = ["EAST", "WEST", "NORTH", "SOUTH"]
    slots = [f"Q{q}_{c}" for q in QUINTANTS for c in CARDINALS]
    p1_scores, p1_meta = defaultdict(dict), defaultdict(dict)


    for key, row in p1_inter_best.items():
        q, card, sk = key
        slot = f"Q{q}_{card}"
        score = brightness_score(best_vapp_for_star(sk))
        p1_scores[slot][sk] = score
        p1_meta[slot][sk] = {"Star": row["Star"], "Source": "INTER-QUINTANT (Phase 1)", "Score": score, "V_App": best_vapp_for_star(sk)}


    for key, row in p1_general_pool.items():
        q, card, sk = key
        slot = f"Q{q}_{card}"
        if sk not in p1_scores[slot]:
            score = brightness_score(best_vapp_for_star(sk))
            p1_scores[slot][sk] = score
            p1_meta[slot][sk] = {"Star": row["Star"], "Source": "PANORAMIC (Phase 1)", "Score": score, "V_App": best_vapp_for_star(sk)}


    p1_candidates = sorted(list(set(k for s in slots for k in p1_scores[s].keys())))
    p1_matrix = np.full((len(slots), len(p1_candidates)), -1e9, dtype=float)


    for i, slot in enumerate(slots):
        for j, sk in enumerate(p1_candidates):
            if sk in p1_scores[slot]:
                p1_matrix[i, j] = p1_scores[slot][sk]


    row_ind, col_ind = linear_sum_assignment(-p1_matrix)
    locked_assignments, locked_stars = {}, set()


    for i, j in zip(row_ind, col_ind):
        slot = slots[i]
        sk = p1_candidates[j]
        if p1_matrix[i, j] <= -1e8: continue
        if "EAST" in slot or "WEST" in slot:
            meta = p1_meta[slot][sk]
            locked_assignments[slot] = {
                "Star": meta["Star"], "StarKey": sk, "Score": p1_matrix[i, j],
                "Source": meta["Source"], "V_App": meta["V_App"]
            }
            locked_stars.add(sk)


    print(f"✓ Locked {len(locked_assignments)} EAST/WEST Inter-Quintant Anchors.")


    # Phase 2: NORTH/SOUTH / VACANT LOCAL FALLBACK
    vacant_slots = [s for s in slots if s not in locked_assignments]


    p2_local_candidates = defaultdict(list)
    for _, row in panoramic_df.iterrows():
        q, sec, sk = int(row["Quintant"]), row["Sector"], row["StarKey"]
        for card, smap in SECTOR_FALLBACK.items():
            if sec in smap:
                p2_local_candidates[(q, card, int(row["Epoch"]))].append({
                    "Star": row["Star"], "StarKey": sk, "V_App": float(row["V_App_Culm"])
                })


    def fallback_global_score(q, cardinal, star_key):
        records = []
        for ep in range(1, n_epochs + 1):
            for c in p2_local_candidates.get((q, cardinal, ep), []):
                if c["StarKey"] == star_key:
                    records.append(brightness_score(c["V_App"]))
        if not records: return None
        return float(np.mean(records))


    p2_fallback_scores, p2_valid_stars = {}, set()
    for slot in vacant_slots:
        q = int(re.search(r'Q(\d+)', slot).group(1))
        card = slot.split("_")[1]
        valid_cands_for_slot = set(c["StarKey"] for ep in range(1, n_epochs+1) for c in p2_local_candidates.get((q, card, ep), []))
        valid_cands_for_slot -= locked_stars
        for sk in valid_cands_for_slot:
            score = fallback_global_score(q, card, sk)
            if score is not None:
                p2_fallback_scores[(slot, sk)] = score
                p2_valid_stars.add(sk)


    p2_stars_list = sorted(list(p2_valid_stars))
    p2_matrix = np.full((len(vacant_slots), len(p2_stars_list)), -1e9, dtype=float)


    for i, slot in enumerate(vacant_slots):
        for j, sk in enumerate(p2_stars_list):
            if (slot, sk) in p2_fallback_scores:
                p2_matrix[i, j] = p2_fallback_scores[(slot, sk)]


    row_ind2, col_ind2 = linear_sum_assignment(-p2_matrix)
    phase2_assignments = {}
    for i, j in zip(row_ind2, col_ind2):
        slot = vacant_slots[i]
        sk = p2_stars_list[j]
        if p2_matrix[i, j] <= -1e8: continue
        star_name = next(c["Star"] for ep in range(1, n_epochs+1) for c in p2_local_candidates.get((int(slot[1]), slot.split('_')[1], ep), []) if c["StarKey"] == sk)
        phase2_assignments[slot] = {
            "Star": star_name, "StarKey": sk, "Score": p2_matrix[i, j],
            "Source": "LOCAL FALLBACK (Phase 2)", "V_App": best_vapp_for_star(sk)
        }


    final_selection = {**locked_assignments, **phase2_assignments}


    print("\n" + "=" * 110)
    print("FINAL 20 INTEGRATED UNIQUE ASTRONOMICAL ANCHORS")
    print("=" * 110)
    print(f"{'SLOT':<12}{'STAR':<20}{'SOURCE':<30}{'SCORE':>10}{'V_APP':>10}")
    print("-" * 110)


    for slot in slots:
        if slot in final_selection:
            x = final_selection[slot]
            print(f"{slot:<12}{x['Star']:<20}{x['Source']:<30}{x['Score']:>10.4f}{x['V_App']:>10.2f}")
        else:
            print(f"{slot:<12}{'VACANT':<20}")


    selected_keys = [x["StarKey"] for x in final_selection.values()]
    if len(selected_keys) != len(set(selected_keys)):
        raise RuntimeError(f"UNIQUENESS FAILURE! Duplicates found: {[k for k in set(selected_keys) if selected_keys.count(k)>1]}")

    # -------------------------------------------------------------
    try:
        print("Attempt 2: Fetching via Astroquery (Harvard mirror)...")
        from astroquery.vizier import Vizier, vizier_conf
        
        # Set global mirror and extended timeout
        vizier_conf.server = "vizier.cfa.harvard.edu"  # Alternative: "vizier.nao.ac.jp"
        
        v = Vizier(
            columns=cols,
            column_filters={"Vmag": f"<{mag_limit}"},
            row_limit=-1,
            timeout=120
        )
        catalog = v.get_catalogs("I/239/hip_main")[0]
        df = catalog.to_pandas()
        print("-> Success via Astroquery mirror!")
        return df
    except Exception as e:
        print(f"-> Attempt 2 failed: {e}")


    # -------------------------------------------------------------
    # Attempt 3: Direct CDS ASU-TSV Stream (Robust Parser)
    # -------------------------------------------------------------
    try:
        print("Attempt 3: Fetching via Direct CDS ASU-TSV stream...")
        col_str = ",".join(cols)
        url = f"https://cdsarc.cds.unistra.fr/viz-bin/asu-tsv?-source=I/239/hip_main&-out={col_str}&Vmag=<{mag_limit}&-out.max=unlimited"
        
        resp = requests.get(url, timeout=30)
        resp.raise_for_status()


        # Remove comment lines
        lines = [line for line in resp.text.splitlines() if not line.startswith("#") and line.strip()]
        
        # Parse TSV dynamically without hardcoded skiprows
        df = pd.read_csv(io.StringIO("\n".join(lines)), sep="|", skipinitialspace=True)
        df.columns = df.columns.str.strip()


        # Coerce HIP to numeric to automatically discard unit/divider rows
        df["HIP"] = pd.to_numeric(df["HIP"], errors="coerce")
        df = df.dropna(subset=["HIP"]).copy()
        df["HIP"] = df["HIP"].astype(int)
        
        # Ensure only requested columns are returned
        df = df[cols].copy()
        print("-> Success via ASU-TSV Stream!")
        return df
    except Exception as e:
        raise RuntimeError(f"All 3 catalog fetch endpoints failed. Last error: {e}")
